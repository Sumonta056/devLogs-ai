Shared settings starting point

Copy this into .claude/settings.json on day one. It covers every item in section 6 that both audited projects failed. Adjust the allow list to the project's own build tools, then run the full build and test suite once with the sandbox on.

{
  "permissions": {
    "deny": [
      "Read(./.env)",
      "Read(./.env.*)",
      "Read(./**/.env*)",
      "Read(./**/*.pem)",
      "Read(./**/*.key)",
      "Read(./**/credentials*)",
      "Read(./**/secrets*)"
    ]
  },
  "sandbox": {
    "enabled": true,
    "failIfUnavailable": true
  },
  "disableSkillShellExecution": true
}

And this pattern for .mcp.json, so no credential is ever written into a tracked file:

{
  "mcpServers": {
    "mongodb": {
      "command": "npx",
      "args": ["-y", "mongodb-mcp-server"],
      "env": { "MDB_MCP_CONNECTION_STRING": "${MONGODB_URI}" }
    }
  }
}

Add this hooks block to the same .claude/settings.json to wire up the project hooks from section 8. The scripts themselves are under each hook item above.

{
  "hooks": {
    "PreToolUse": [
      {
        "matcher": "Read|Edit|MultiEdit|Write|Bash",
        "hooks": [
          { "type": "command", "command": "$CLAUDE_PROJECT_DIR/.claude/hooks/guard-secrets.sh" }
        ]
      },
      {
        "matcher": "Bash",
        "hooks": [
          { "type": "command", "command": "$CLAUDE_PROJECT_DIR/.claude/hooks/block-dangerous.sh" },
          { "type": "command", "command": "$CLAUDE_PROJECT_DIR/.claude/hooks/commit-gate.sh" }
        ]
      }
    ],
    "PostToolUse": [
      {
        "matcher": "Edit|MultiEdit|Write",
        "hooks": [
          { "type": "command", "command": "$CLAUDE_PROJECT_DIR/.claude/hooks/format-changed.sh" },
          { "type": "command", "command": "$CLAUDE_PROJECT_DIR/.claude/hooks/test-changed.sh", "timeout": 120 }
        ]
      }
    ]
  }
}
